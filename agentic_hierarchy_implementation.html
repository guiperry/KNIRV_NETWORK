1. Overview

Each DVE is auto-provisioned with a dedicated KNIRVAGENT supervisor process on DVE creation. This spec extends that architecture so the KNIRVAGENT supervisor can spawn, monitor, and proxy arbitrary inner agent CLI tools (e.g. claude, aider, codex) inside the DVE workspace. The frontend gains a multi-pane terminal workspace where each inner agent session occupies its own tab alongside the supervisor tab.

Design Principles

  • All internal communication uses Unix sockets — no TCP between services.
  • KNIRVGATEWAY (:8888) is the sole public-facing proxy. All browser-reachable endpoints must pass through it.
  • New inner agent library code belongs in packages/KNIRVAGENT/pkg/inner/ and packages/KNIRVAGENT/pkg/server/ — the binary's own module (github.com/knirvcorp/knirvagent).
  • The KNIRVAGENT binary's HTTP server (serverCmd() in cmd/knirvagent/main.go) is extended with new /api/inner/* endpoints, still served over the existing per-DVE Unix socket.
  • KNIRVSERVER manages connections to KNIRVAGENT via packages/KNIRVSERVER/pkg/knirvagent/manager.go — new proxy helper methods are added there to call the inner endpoints.
  • Every byte written by an inner agent is tee'd: one copy to a supervisor log file, one to subscriber channels (WebSocket streams).

4. Unix Socket Topology

No TCP between internal services. All cross-service hops are Unix socket dials. KNIRVGATEWAY is the only TCP listener the browser talks to.

Socket PathOwnerClients
/var/lib/knirvserver/sockets/backend.sock KNIRVSERVER backend KNIRVGATEWAY (for /api/v1/* and NEW /ws/*)
/var/lib/knirvserver/sockets/agent-{dveID}.sock KNIRVAGENT subprocess (one per DVE) KNIRVSERVER backend handlers via AgentManager.GetSocketPathForDVE()
/var/lib/knirvserver/sockets/chain.sock KNIRVCHAIN KNIRVSERVER backend
/var/lib/knirvserver/sockets/graph.sock KNIRVGRAPH KNIRVSERVER backend
Inner Sessions — No New Sockets

Inner agent sessions do not get their own Unix sockets. All inner session traffic flows through the existing agent-{dveID}.sock. The KNIRVAGENT binary's HTTP mux gains new /api/inner/* routes served on the same socket. The KNIRVSERVER backend's dve_inner_agent_ws_handler.go dials that socket and streams PTY output back to the browser over a WebSocket.

5.1 packages/KNIRVAGENT/pkg/inner/session.go NEW

package inner

import (
    "io"
    "os"
    "os/exec"
    "sync"
    "time"
)

type SessionStatus string

const (
    StatusRunning SessionStatus = "running"
    StatusExited  SessionStatus = "exited"
)

type InnerAgentSession struct {
    ID           string
    ToolName     string
    Cmd          *exec.Cmd
    PTY          *os.File       // PTY master fd — write = keystrokes, read = output
    Status       SessionStatus
    StartedAt    time.Time
    WorkspaceDir string
    DVEID        string

    supervisorLog io.Writer    // written by broadcast(); every byte tee'd here

    mu          sync.RWMutex
    subscribers []chan []byte  // one channel per active /stream consumer
    done        chan struct{}
}

// Subscribe returns a channel that receives every PTY output chunk.
// Closed when the session exits. Caller must drain it.
func (s *InnerAgentSession) Subscribe() chan []byte {
    ch := make(chan []byte, 256)
    s.mu.Lock()
    s.subscribers = append(s.subscribers, ch)
    s.mu.Unlock()
    return ch
}

func (s *InnerAgentSession) Unsubscribe(ch chan []byte) {
    s.mu.Lock()
    defer s.mu.Unlock()
    for i, sub := range s.subscribers {
        if sub == ch {
            s.subscribers = append(s.subscribers[:i], s.subscribers[i+1:]...)
            return
        }
    }
}

// broadcast tees a PTY output chunk: supervisor log + all subscriber channels.
func (s *InnerAgentSession) broadcast(chunk []byte) {
    s.supervisorLog.Write(chunk)
    s.mu.RLock()
    defer s.mu.RUnlock()
    for _, ch := range s.subscribers {
        select {
        case ch <- append([]byte(nil), chunk...):
        default: // drop if subscriber is slow
        }
    }
}

// PumpOutput reads from the PTY master and fans out until the process exits.
// Run as a goroutine immediately after Spawn.
func (s *InnerAgentSession) PumpOutput() {
    buf := make([]byte, 4096)
    for {
        n, err := s.PTY.Read(buf)
        if n > 0 {
            s.broadcast(buf[:n])
        }
        if err != nil { break }
    }
    s.mu.Lock()
    s.Status = StatusExited
    for _, ch := range s.subscribers { close(ch) }
    s.subscribers = nil
    s.mu.Unlock()
    close(s.done)
}

5.2 packages/KNIRVAGENT/pkg/inner/manager.go NEW

package inner

import (
    "fmt"
    "os"
    "sync"
    "time"

    "github.com/creack/pty"
    "github.com/google/uuid"
)

type InnerAgentManager struct {
    mu           sync.RWMutex
    sessions     map[string]*InnerAgentSession
    catalog      *ToolCatalog
    workspaceDir string
    dveID        string
    logDir       string  // {workspaceDir}/.supervisor-logs/
}

func NewInnerAgentManager(dveID, workspaceDir string) *InnerAgentManager {
    logDir := workspaceDir + "/.supervisor-logs"
    os.MkdirAll(logDir, 0755)
    return &InnerAgentManager{
        sessions:     make(map[string]*InnerAgentSession),
        catalog:      DefaultCatalog(),
        workspaceDir: workspaceDir,
        dveID:        dveID,
        logDir:       logDir,
    }
}

// Spawn launches an inner agent CLI tool with a PTY, wires supervisor tee, starts output pump.
func (m *InnerAgentManager) Spawn(toolName string, extraEnv []string) (*InnerAgentSession, error) {
    tool, ok := m.catalog.Lookup(toolName)
    if !ok {
        return nil, fmt.Errorf("unknown tool: %s", toolName)
    }
    if !tool.IsInstalled() {
        return nil, fmt.Errorf("tool %s not installed; call Install first", toolName)
    }

    sessionID := uuid.NewString()
    logFile, err := os.Create(fmt.Sprintf("%s/%s.log", m.logDir, sessionID))
    if err != nil { return nil, err }

    cmd := tool.BuildCmd(m.workspaceDir, extraEnv)
    DefaultSpawnHooks(cmd, m.dveID, sessionID)

    ptmx, err := pty.Start(cmd)
    if err != nil { logFile.Close(); return nil, fmt.Errorf("pty.Start: %w", err) }

    sess := &InnerAgentSession{
        ID:            sessionID,
        ToolName:      toolName,
        Cmd:           cmd,
        PTY:           ptmx,
        Status:        StatusRunning,
        StartedAt:     time.Now(),
        WorkspaceDir:  m.workspaceDir,
        DVEID:         m.dveID,
        supervisorLog: logFile,
        done:          make(chan struct{}),
    }

    m.mu.Lock()
    m.sessions[sessionID] = sess
    m.mu.Unlock()

    go sess.PumpOutput()
    return sess, nil
}

// SendInput writes raw bytes (keystrokes) to the session's PTY master.
func (m *InnerAgentManager) SendInput(sessionID string, data []byte) error {
    sess := m.get(sessionID)
    if sess == nil { return fmt.Errorf("session %s not found", sessionID) }
    _, err := sess.PTY.Write(data)
    return err
}

// Install runs the tool's install command inside the DVE workspace directory.
func (m *InnerAgentManager) Install(toolName string) error {
    tool, ok := m.catalog.Lookup(toolName)
    if !ok { return fmt.Errorf("unknown tool: %s", toolName) }
    return tool.Install(m.workspaceDir)
}

func (m *InnerAgentManager) Terminate(sessionID string) error {
    sess := m.get(sessionID)
    if sess == nil { return nil }
    if sess.Cmd != nil && sess.Cmd.Process != nil {
        sess.Cmd.Process.Kill()
    }
    sess.PTY.Close()
    m.mu.Lock()
    delete(m.sessions, sessionID)
    m.mu.Unlock()
    return nil
}

func (m *InnerAgentManager) GetSession(id string) (*InnerAgentSession, error) {
    sess := m.get(id)
    if sess == nil { return nil, fmt.Errorf("session %s not found", id) }
    return sess, nil
}

func (m *InnerAgentManager) ListSessions() []*InnerAgentSession {
    m.mu.RLock()
    defer m.mu.RUnlock()
    out := make([]*InnerAgentSession, 0, len(m.sessions))
    for _, s := range m.sessions { out = append(out, s) }
    return out
}

func (m *InnerAgentManager) ListTools() []Tool { return m.catalog.All() }

func (m *InnerAgentManager) get(id string) *InnerAgentSession {
    m.mu.RLock()
    defer m.mu.RUnlock()
    return m.sessions[id]
}

package inner

import (
    "os"
    "os/exec"
)

type Tool struct {
    Name        string
    Binary      string   // executable name resolved via $PATH
    DefaultArgs []string
    InstallCmd  []string // run once to install
}

func (t *Tool) IsInstalled() bool { _, err := exec.LookPath(t.Binary); return err == nil }

func (t *Tool) Install(workdir string) error {
    cmd := exec.Command(t.InstallCmd[0], t.InstallCmd[1:]...)
    cmd.Dir = workdir
    cmd.Stdout, cmd.Stderr = os.Stdout, os.Stderr
    return cmd.Run()
}

func (t *Tool) BuildCmd(workdir string, extra []string) *exec.Cmd {
    cmd := exec.Command(t.Binary, t.DefaultArgs...)
    cmd.Dir = workdir
    cmd.Env = append(os.Environ(), extra...)
    return cmd
}

type ToolCatalog struct { tools map[string]*Tool }

func (c *ToolCatalog) Lookup(name string) (*Tool, bool) { t, ok := c.tools[name]; return t, ok }
func (c *ToolCatalog) All() []Tool {
    out := make([]Tool, 0, len(c.tools))
    for _, t := range c.tools { out = append(out, *t) }
    return out
}

5.5 packages/KNIRVAGENT/pkg/server/inner_routes.go NEW

Mounts /api/inner/* onto the http.ServeMux already used in serverCmd(). The /api/inner/{sessionId}/stream endpoint is a long-lived chunked response that KNIRVSERVER's WebSocket handler streams from. 

package server

import (
    "encoding/json"
    "io"
    "net/http"
    "os"
    "strings"

    "github.com/knirvcorp/knirvagent/pkg/inner"
)

// RegisterInnerRoutes attaches all /api/inner/* handlers to an http.ServeMux.
// Called from cmd/knirvagent/main.go serverCmd() after existing routes are registered.
func RegisterInnerRoutes(mux *http.ServeMux, mgr *inner.InnerAgentManager) {
    mux.HandleFunc("/api/inner/spawn",    handleSpawn(mgr))
    mux.HandleFunc("/api/inner/install",  handleInstall(mgr))
    mux.HandleFunc("/api/inner/sessions", handleListSessions(mgr))
    mux.HandleFunc("/api/inner/tools",    handleListTools(mgr))
    mux.HandleFunc("/api/inner/",         handleSessionRouter(mgr)) // routes /api/inner/{id}/*
}

// handleSessionRouter dispatches /api/inner/{sessionId}/input, /stream, /log and DELETE /api/inner/{sessionId}.
func handleSessionRouter(mgr *inner.InnerAgentManager) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        // path: /api/inner/{sessionId}[/action]
        rest := strings.TrimPrefix(r.URL.Path, "/api/inner/")
        parts := strings.SplitN(rest, "/", 2)
        sessionID := parts[0]
        action := ""
        if len(parts) == 2 { action = parts[1] }

        switch {
        case r.Method == "DELETE" && action == "":
            mgr.Terminate(sessionID)
            w.WriteHeader(http.StatusNoContent)
        case action == "input" && r.Method == "POST":
            data, _ := io.ReadAll(io.LimitReader(r.Body, 4096))
            mgr.SendInput(sessionID, data)
            w.WriteHeader(http.StatusNoContent)
        case action == "stream" && r.Method == "GET":
            handleStream(mgr, sessionID, w, r)
        case action == "log" && r.Method == "GET":
            handleGetLog(mgr, sessionID, w)
        default:
            http.NotFound(w, r)
        }
    }
}

// handleStream is a long-lived chunked response; subscriber channel drains PTY output.
func handleStream(mgr *inner.InnerAgentManager, sessionID string, w http.ResponseWriter, r *http.Request) {
    sess, err := mgr.GetSession(sessionID)
    if err != nil { http.Error(w, err.Error(), http.StatusNotFound); return }

    w.Header().Set("Content-Type", "application/octet-stream")
    w.Header().Set("Transfer-Encoding", "chunked")
    w.Header().Set("X-Accel-Buffering", "no")
    w.WriteHeader(http.StatusOK)

    flusher, _ := w.(http.Flusher)
    ch := sess.Subscribe()
    defer sess.Unsubscribe(ch)

    for {
        select {
        case chunk, ok := <-ch:
            if !ok { return }
            if _, err := w.Write(chunk); err != nil { return }
            if flusher != nil { flusher.Flush() }
        case <-r.Context().Done():
            return
        }
    }
}

func handleGetLog(mgr *inner.InnerAgentManager, sessionID string, w http.ResponseWriter) {
    sess, err := mgr.GetSession(sessionID)
    if err != nil { http.Error(w, err.Error(), http.StatusNotFound); return }
    logPath := sess.WorkspaceDir + "/.supervisor-logs/" + sessionID + ".log"
    data, err := os.ReadFile(logPath)
    if err != nil { http.Error(w, "log not found", http.StatusNotFound); return }
    w.Header().Set("Content-Type", "text/plain")
    w.Write(data)
}

// Add a package-level helper that builds an http.Client dialing a Unix socket path.
// Shared by all new manager proxy methods.
func unixHTTPClient(socketPath string) *http.Client {
    return &http.Client{
        Transport: &http.Transport{
            DialContext: func(ctx context.Context, _, _ string) (net.Conn, error) {
                return net.DialTimeout("unix", socketPath, 5*time.Second)
            },
        },
    }
}

// ── Inner agent proxy methods ─────────────────────────────────────────────────

// SpawnInnerSession calls POST /api/inner/spawn on the DVE's KNIRVAGENT socket.
func (am *AgentManager) SpawnInnerSession(ctx context.Context, dveID, toolName string) (map[string]interface{}, error) {
    socketPath, err := am.getSocketPath(dveID)
    if err != nil { return nil, err }
    body, _ := json.Marshal(map[string]string{"tool": toolName})
    req, _ := http.NewRequestWithContext(ctx, "POST", "http://localhost/api/inner/spawn", bytes.NewReader(body))
    req.Header.Set("Content-Type", "application/json")
    resp, err := unixHTTPClient(socketPath).Do(req)
    if err != nil { return nil, err }
    defer resp.Body.Close()
    var result map[string]interface{}
    json.NewDecoder(resp.Body).Decode(&result)
    return result, nil
}

// GetInnerSessionStream opens a long-lived chunked stream for inner session PTY output.
// Returns an io.ReadCloser the caller must close. Used by DVEInnerAgentWSHandler.
func (am *AgentManager) GetInnerSessionStream(ctx context.Context, dveID, sessionID string) (io.ReadCloser, error) {
    socketPath, err := am.getSocketPath(dveID)
    if err != nil { return nil, err }
    url := "http://localhost/api/inner/" + sessionID + "/stream"
    req, _ := http.NewRequestWithContext(ctx, "GET", url, nil)
    resp, err := unixHTTPClient(socketPath).Do(req)
    if err != nil { return nil, err }
    return resp.Body, nil
}

7.1 backend/internal/web/dve_inner_agent_ws_handler.go NEW

Upgrades browser WebSocket at /ws/dve/{nodeId}/inner/{sessionId}. Uses AgentManager.GetInnerSessionStream() to open the chunked PTY stream from agent-{dveID}.sock, then fans raw bytes to the browser as {"type":"data","data":"..."} JSON frames. Inbound browser keystrokes are forwarded to AgentManager.SendInnerInput(). 

package web

// Route registered in api_router.go:
// r.HandleFunc("/ws/dve/{nodeId}/inner/{sessionId}", innerWSHandler.HandleWebSocket)

func (h *DVEInnerAgentWSHandler) HandleWebSocket(w http.ResponseWriter, r *http.Request) {
    vars := mux.Vars(r)
    nodeID, sessionID := vars["nodeId"], vars["sessionId"]

    if _, err := h.agentManager.GetSocketPathForDVE(nodeID); err != nil {
        http.Error(w, "no agent for DVE", http.StatusNotFound); return
    }

    conn, err := upgrader.Upgrade(w, r, nil)
    if err != nil { return }
    defer conn.Close()

    streamCtx, cancel := context.WithCancel(r.Context())
    defer cancel()

    stream, err := h.agentManager.GetInnerSessionStream(streamCtx, nodeID, sessionID)
    if err != nil {
        conn.WriteJSON(map[string]string{"type": "error", "data": err.Error()})
        return
    }
    defer stream.Close()

    // goroutine: PTY stream → WebSocket send
    go func() {
        buf := make([]byte, 4096)
        for {
            n, err := stream.Read(buf)
            if n > 0 {
                conn.SetWriteDeadline(time.Now().Add(10 * time.Second))
                conn.WriteJSON(map[string]string{"type": "data", "data": string(buf[:n])})
            }
            if err != nil { cancel(); return }
        }
    }()

    // main loop: browser input → SendInnerInput
    for {
        _, raw, err := conn.ReadMessage()
        if err != nil { break }
        var msg struct{ Type, Data string }
        if json.Unmarshal(raw, &msg) == nil && msg.Type == "input" {
            inputCtx, c := context.WithTimeout(context.Background(), 5*time.Second)
            h.agentManager.SendInnerInput(inputCtx, nodeID, sessionID, []byte(msg.Data))
            c()
        }
    }
}

Critical Pre-Requisite — WebSocket Proxy Gap

packages/KNIRVGATEWAY/internal/server/server.go proxies /api/v1/* to backend.sock but has no rule for /ws/*. Browser WebSocket connections to wss://host:8888/ws/dve/{nodeId}/agent (supervisor) and the new wss://host:8888/ws/dve/{nodeId}/inner/{sessionId} are silently dropped. This must be fixed first.

packages/KNIRVGATEWAY/internal/server/server.go MODIFIED

Standard httputil.ReverseProxy strips Upgrade and Connection headers, breaking WebSockets. The fix uses raw TCP hijacking to tunnel the WebSocket framing through the Unix socket without modification. 

// In setupRoutes() — AFTER the existing /api/v1/ proxy rule:

// ── existing ──────────────────────────────────────────────────────────────
backendProxy := newSocketProxy(cfg.BackendSocketPath, "")
r.PathPrefix("/api/v1/").Handler(backendProxy)

// ── ADD: WebSocket tunnel to backend.sock ─────────────────────────────────
// httputil.ReverseProxy cannot handle WebSocket upgrades — use raw TCP tunnel.
r.PathPrefix("/ws/").HandlerFunc(newWebSocketTunnel(cfg.BackendSocketPath))

// ── new helper function to add to server.go ───────────────────────────────
func newWebSocketTunnel(socketPath string) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        backendConn, err := net.Dial("unix", socketPath)
        if err != nil {
            http.Error(w, "backend unavailable", http.StatusBadGateway)
            return
        }
        defer backendConn.Close()

        hijacker, ok := w.(http.Hijacker)
        if !ok { http.Error(w, "hijacking not supported", http.StatusInternalServerError); return }
        clientConn, buf, err := hijacker.Hijack()
        if err != nil { return }
        defer clientConn.Close()

        // Replay the original HTTP request to the backend.
        r.Write(backendConn)
        if n := buf.Reader().Buffered(); n > 0 {
            io.CopyN(backendConn, buf, int64(n))
        }

        // Bidirectional copy — both sides run concurrently; first done cancels both.
        done := make(chan struct{}, 2)
        go func() { io.Copy(backendConn, clientConn); done <- struct{}{}{}() }()
        go func() { io.Copy(clientConn, backendConn); done <- struct{}{}{}() }()
        <-done
    }
}
